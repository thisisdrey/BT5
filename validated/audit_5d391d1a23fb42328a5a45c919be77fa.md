Based on my investigation, this repository is the Go standard library / distribution itself (a fork), not an application that consumes untrusted PDFs or pickle files. I searched for the closest analogous patterns to the reported CVE-2025-64512 primitive — insecure deserialization combined with attacker-controlled file path/directory traversal leading to code execution:

- `encoding/gob` — its `Decoder` explicitly documents that it is **not hardened against adversarial input** and is **out of scope of the Go security policy**: [1](#0-0)  Even so, `gob.Decode` only invokes `GobDecode`/`UnmarshalBinary`/`UnmarshalText` on types that the *host program* has already registered/compiled into its type table — it cannot execute attacker-supplied code the way `pickle.loads()` can, since Go's static compilation model precludes deserialization defining new executable behavior. [2](#0-1) 
- `archive/zip` — explicitly defends against the analogous "zip slip" path-traversal-via-crafted-archive bug class (CVE-2021-27919 pattern) via `filepath.IsLocal` checks and `ErrInsecurePath`, gated by `GODEBUG=zipinsecurepath`. [3](#0-2)  This is a fixed, defended path, not a live vulnerability.
- `cmd/go/internal/modfetch` — the module cache/download path already validates that zip entries are prefixed by `module@version/` before extraction, preventing an attacker-controlled module zip from writing outside its designated directory. [4](#0-3) 
- `plugin.Open` — this does dynamically load and execute arbitrary native code (`dlopen`), but it requires the *path to the .so file* to already be a trusted, locally-supplied argument controlled by the host program, not attacker-supplied data embedded in a document format; the package's own docs call out that "bugs in applications that load plugins could be exploited by an attacker to load dangerous or untrusted libraries" as an application-level risk, not a stdlib defect. [5](#0-4) [6](#0-5) 

None of these constitute an unauthenticated/untrusted-input-triggered arbitrary-code-execution primitive analogous to pdfminer's `CMapDB._load_data()` pickle-from-attacker-specified-path bug reachable through a normal victim workflow (e.g., parsing an untrusted document/module) in this codebase. The gob package's danger is explicitly documented and requires the victim program to already register attacker-influenceable types — that's an application misuse case, not a stdlib vulnerability, and is explicitly out of Go's security policy scope per its own documentation.

// decodeGobDecoder decodes something implementing the GobDecoder interface.
// The data is encoded as a byte slice.
func (dec *Decoder) decodeGobDecoder(ut *userTypeInfo, state *decoderState, value reflect.Value) {
	// Read the bytes for the value.
	n, ok := state.getLength()
	if !ok {
		errorf("GobDecoder: length too large for buffer")
	}
	b := state.b.Bytes()
	if len(b) < n {
		errorf("GobDecoder: invalid data length %d: exceeds input size %d", n, len(b))
	}
	b = b[:n]
	state.b.Drop(n)
	var err error
	// We know it's one of these.
	switch ut.externalDec {
	case xGob:
		gobDecoder, _ := reflect.TypeAssert[GobDecoder](value)
		err = gobDecoder.GobDecode(b)
	case xBinary:
		binaryUnmarshaler, _ := reflect.TypeAssert[encoding.BinaryUnmarshaler](value)
		err = binaryUnmarshaler.UnmarshalBinary(b)
	case xText:
		textUnmarshaler, _ := reflect.TypeAssert[encoding.TextUnmarshaler](value)
		err = textUnmarshaler.UnmarshalText(b)
	}
	if err != nil {
		error_(err)
	}
}

func open(name string) (*Plugin, error) {
	cPath := make([]byte, C.PATH_MAX+1)
	cRelName := make([]byte, len(name)+1)
	copy(cRelName, name)
	if C.realpath(
		(*C.char)(unsafe.Pointer(&cRelName[0])),
		(*C.char)(unsafe.Pointer(&cPath[0]))) == nil {
		return nil, errors.New(`plugin.Open("` + name + `"): realpath failed`)
	}

	filepath := C.GoString((*C.char)(unsafe.Pointer(&cPath[0])))

	pluginsMu.Lock()
	if p := plugins[filepath]; p != nil {
		pluginsMu.Unlock()
		if p.err != "" {
			return nil, errors.New(`plugin.Open("` + name + `"): ` + p.err + ` (previous failure)`)
		}
		<-p.loaded
		return p, nil
	}

	h := C.pluginOpen((*C.char)(unsafe.Pointer(&cPath[0])), &cErr)

		pluginsMu.Unlock()
		return nil, errors.New(`plugin.Open("` + name + `"): ` + C.GoString(cErr))
