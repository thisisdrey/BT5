**Findings from investigating the repo (`Kohvert/go--001`, a fork of the Go toolchain source):**

The CVE-2024-1873 report describes an *application-level* vulnerability in `parisneo/lollms-webui`: an unauthenticated `/select_database` HTTP endpoint that passes attacker-controlled absolute paths directly into database/directory operations, with no path validation. This is an application bug, not a language/toolchain bug.

I looked for the closest analogous pattern in the Go toolchain — places where untrusted external input (module paths, archive entry names, import paths) flows into filesystem directory-creation or extraction sinks — and found that Go already implements systematic defenses at every such boundary:

- Module cache paths are never taken raw; they go through `module.EscapePath`/`module.CheckImportPath` before being joined into `GOMODCACHE` paths in `cacheDir`, `CachePath`, and `DownloadDir`. [1](#0-0) [2](#0-1) 
- VCS checkout work directories are named by a SHA-256 hash of type+name rather than the untrusted name itself, explicitly to avoid "security problems" from nested/traversing checkouts. [3](#0-2) 
- Module zip extraction (`unzip` → `modzip.Unzip`) operates on a `dir` computed via the same escaped/validated path, and `archive/zip` itself rejects `../`-style entries via `ErrInsecurePath` unless explicitly opted out with `GODEBUG=zipinsecurepath=0`. [4](#0-3) [5](#0-4) 
- Import paths (both compiler-level and module-level) are validated by `checkImportPath`/`module.CheckImportPath` to reject absolute paths, `..`, control characters, etc. [6](#0-5) [7](#0-6) 
- The newer `os.Root` API provides explicit symlink/`..`-escape detection (`checkPathEscapes`, `rootMkdirAll`) specifically to make directory creation/traversal into an untrusted directory tree safe. [8](#0-7) [9](#0-8) 

None of these represent an unauthenticated, reachable Go-toolchain entry point with a missing check analogous to `/select_database`'s absent path validation — every comparable sink (module cache dir creation, zip extraction, VCS workdir creation, import path resolution) already validates or escapes the untrusted string before it reaches `os.MkdirAll`/`os.Mkdir`/extraction. This is a well-hardened area of Go (it was the subject of past fixes like CVE-2021-27919), and I found no unfixed instance matching the reported bug class.

// WorkDir returns the name of the cached work directory to use for the
// given repository type and name.
func WorkDir(ctx context.Context, typ, name string) (dir, lockfile string, err error) {
	if cfg.GOMODCACHE == "" {
		return "", "", fmt.Errorf("neither GOPATH nor GOMODCACHE are set")
	}

	// We name the work directory for the SHA256 hash of the type and name.
	// We intentionally avoid the actual name both because of possible
	// conflicts with valid file system paths and because we want to ensure
	// that one checkout is never nested inside another. That nesting has
	// led to security problems in the past.
	if strings.Contains(typ, ":") {
		return "", "", fmt.Errorf("codehost.WorkDir: type cannot contain colon")
	}
	key := typ + ":" + name
	dir = filepath.Join(cfg.GOMODCACHE, "cache/vcs", fmt.Sprintf("%x", sha256.Sum256([]byte(key))))

func checkImportPath(path string, allowSpace bool) error {
	if path == "" {
		return errors.New("import path is empty")
	}

	if strings.Contains(path, "\x00") {
		return errors.New("import path contains NUL")
	}

	for ri := range base.ReservedImports {
		if path == ri {
			return fmt.Errorf("import path %q is reserved and cannot be used", path)
		}
	}

	for _, r := range path {
		switch {
		case r == utf8.RuneError:
			return fmt.Errorf("import path contains invalid UTF-8 sequence: %q", path)
		case r < 0x20 || r == 0x7f:
			return fmt.Errorf("import path contains control character: %q", path)
		case r == '\\':
			return fmt.Errorf("import path contains backslash; use slash: %q", path)
		case !allowSpace && unicode.IsSpace(r):
			return fmt.Errorf("import path contains space character: %q", path)
		case strings.ContainsRune("!\"#$%&'()*,:;<=>?[]^`{|}", r):
			return fmt.Errorf("import path contains invalid character '%c': %q", r, path)
		}
	}

	return nil
}

func rootMkdirAll(r *Root, fullname string, perm FileMode) error {
	// doInRoot opens each path element in turn.
	//
	// openDirFunc opens all but the last path component.
	// The usual default openDirFunc just opens directories with O_DIRECTORY.
	// We replace it here with one that creates missing directories along the way.
	openDirFunc := func(parent sysfdType, name string) (sysfdType, error) {
		for try := range 2 {
			fd, err := rootOpenDir(parent, name)
			switch err.(type) {
			case nil, errSymlink:
				return fd, err
			}
			if try > 0 || !IsNotExist(err) {
				return 0, &PathError{Op: "openat", Err: err}
			}
			// Try again on EEXIST, because the directory may have been created
			// by another process or thread between the rootOpenDir and mkdirat calls.
			if err := mkdirat(parent, name, perm); err != nil && err != syscall.EEXIST {
				return 0, &PathError{Op: "mkdirat", Err: err}
			}
		}
		panic("unreachable")
	}
	// openLastComponentFunc opens the last path component.
	openLastComponentFunc := func(parent sysfdType, name string, endsInSlash bool) (struct{}, error) {
		err := mkdirat(parent, name, perm)
		if err == syscall.EEXIST {
			mode, e := modeAt(parent, name)
			if e == nil {
				if mode.IsDir() {
					// The target of MkdirAll is an existing directory.
					err = nil
				} else if mode&ModeSymlink != 0 {
					// The target of MkdirAll is a symlink.
					// For consistency with os.MkdirAll,
					// succeed if the link resolves to a directory.
					// We don't return errSymlink here, because we don't
					// want to create the link target if it doesn't exist.
					fi, e := r.Stat(fullname)
					if e == nil && fi.Mode().IsDir() {
						err = nil
					} else if e == nil {
						err = syscall.ENOTDIR
					} else if !IsNotExist(e) {
						// EPERM, ELOOP, etc.,
						// probably more useful than EEXIST.
						err = e
					}
				}
			}
		}
		switch err.(type) {
		case nil, errSymlink:
			return struct{}{}, err
		}
		return struct{}{}, &PathError{Op: "mkdirat", Err: err}
	}
	flags := uint(doInRootCreatingDirectory)
	switch runtime.GOOS {
	case "linux", "windows", "openbsd":
		flags |= doInRootNoHandleTerminalSlash // see rootMkdir
	}
	_, err := doInRoot(r, fullname, flags, openDirFunc, openLastComponentFunc)
	if err != nil {
		if _, ok := err.(*PathError); !ok {
			err = &PathError{Op: "mkdirat", Path: fullname, Err: err}
		}
	}
	return err
}
