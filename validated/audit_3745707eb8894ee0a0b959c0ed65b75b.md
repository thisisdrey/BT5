`decode_signer` in `stackslib/src/net/api/getsigner.rs` is simply the client-side HTTP response decoder for the `/v3/signer/:signer_pubkey/:cycle_num` RPC endpoint — it parses a JSON body into a `GetSignerResponse { blocks_signed: u64 }` struct [1](#0-0) . It has no involvement with P2P handshakes, handshake signatures, peer network-id/address binding, StackerDB chunk signing, Atlas attachments, or inventory messages. The request-side parsing (`try_parse_request`) only extracts a public key and reward cycle from the URL path and requires a zero-length body [2](#0-1) , and the handler (`try_handle_request`) merely looks up a signer's block count from chainstate [3](#0-2) . There is no signature verification, peer identity, or network-id equality check anywhere in this file, so the claimed invariant ("peer identity/network bound == the one the signature authenticated") is not applicable to this code path at all.

    fn try_parse_request(
        &mut self,
        preamble: &HttpRequestPreamble,
        captures: &Captures,
        query: Option<&str>,
        _body: &[u8],
    ) -> Result<HttpRequestContents, Error> {
        if preamble.get_content_length() != 0 {
            return Err(Error::DecodeError(
                "Invalid Http request: expected 0-length body".into(),
            ));
        }

        let Some(cycle_num_str) = captures.name("cycle_num") else {
            return Err(Error::DecodeError(
                "Missing in request path: `cycle_num`".into(),
            ));
        };
        let Some(signer_pubkey_str) = captures.name("signer_pubkey") else {
            return Err(Error::DecodeError(
                "Missing in request path: `signer_pubkey`".into(),
            ));
        };

        let signer_pubkey = Secp256k1PublicKey::from_hex(signer_pubkey_str.into())
            .map_err(|e| Error::DecodeError(format!("Failed to signer public key: {e}")))?;

        let cycle_num = u64::from_str_radix(cycle_num_str.into(), 10)
            .map_err(|e| Error::DecodeError(format!("Failed to parse cycle number: {e}")))?;

        self.signer_pubkey = Some(signer_pubkey);
        self.reward_cycle = Some(cycle_num);

        Ok(HttpRequestContents::new().query_string(query))
    }

        let result = node.with_node_state(|_network, _sortdb, chainstate, _mempool, _rpc_args| {

                &chainstate.index_conn(),
                &signer_pubkey,
                reward_cycle,
            )
        });

        let blocks_signed = match result {

            Err(error) => {

                .map_err(NetError::from);
            }
        };

        let preamble = HttpResponsePreamble::ok_json(&preamble);
        let body = HttpResponseContents::try_from_json(&response)?;

impl StacksHttpResponse {
    pub fn decode_signer(self) -> Result<GetSignerResponse, NetError> {
        let contents = self.get_http_payload_ok()?;
        let response_json: serde_json::Value = contents.try_into()?;
        let response: GetSignerResponse = serde_json::from_value(response_json)
            .map_err(|_e| Error::DecodeError("Failed to decode JSON".to_string()))?;
        Ok(response)
    }
