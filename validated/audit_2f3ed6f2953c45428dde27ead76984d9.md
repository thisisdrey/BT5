The SDL CVE describes a heap-based buffer **over-read** in `BlitNtoN`/`SDL_SoftBlit` — a C-level memory-safety bug where pixel blitting reads past the end of a buffer without bounds checking, silently disclosing adjacent heap memory into the output image.

Go's closest analog domain is `image/draw` (pixel blitting, e.g. `drawRGBA`, `drawCopyOver`, `drawNRGBAOver` in `src/image/draw/draw.go`) and the image decoders (`image/gif`, `image/png`, `image/jpeg`). I examined these for the same primitive — an untrusted-input-driven pixel copy/blit that reads outside an allocated buffer without a check. [1](#0-0) [2](#0-1) 

These findings do not support a matching analog:

1. **Different memory-safety model.** Every pixel access in `image/draw` and the image decoders goes through Go slices (e.g. `dst.Pix[i : i+4 : i+4]`, `spix[si : si+4 : si+4]`). Go's runtime enforces bounds checks on every slice/index operation; an out-of-bounds access does not silently read adjacent heap memory (as in C) — it triggers a `runtime error: index out of range` panic via `panicBounds32`/`panicBounds64`. [3](#0-2)  This converts the C bug class (silent heap disclosure) into a safe, recoverable panic, which is a fundamentally different security posture, not the same primitive.

2. **Explicit validation already exists precisely for this bug class.** The `image/gif` decoder already validates palette indices are within range before use, and validates frame bounds against image bounds, specifically to prevent malformed/untrusted pixel data from causing out-of-bounds effects. [4](#0-3) [5](#0-4) 

3. **Blit routines operate on internally-consistent geometry**, not on attacker-controlled raw offsets analogous to SDL's `BlitNtoN`. Source/destination rectangles are intersected and clamped by `DrawMask` before any of the row-copy helper functions (`drawCopySrc`, `drawCopyOver`, `drawNRGBAOver`, etc.) are invoked, so there's no reachable path where an attacker-supplied image causes a read past its own `Pix` slice without first hitting Go's bounds-checked slicing (which would panic, not over-read).

Since the underlying memory-safety guarantee of Go slices structurally prevents the "silent heap over-read" primitive described in the SDL CVE, and the closest candidate code (`image/draw`, `image/gif`) already has both bounds-checked slice access and explicit input validation, there is no reproducible analog matching this bug class in the scoped files.

func drawCopyOver(dst *image.RGBA, r image.Rectangle, src *image.RGBA, sp image.Point) {
	dx, dy := r.Dx(), r.Dy()
	d0 := dst.PixOffset(r.Min.X, r.Min.Y)
	s0 := src.PixOffset(sp.X, sp.Y)
	var (
		ddelta, sdelta int
		i0, i1, idelta int
	)
	if r.Min.Y < sp.Y || r.Min.Y == sp.Y && r.Min.X <= sp.X {
		ddelta = dst.Stride
		sdelta = src.Stride
		i0, i1, idelta = 0, dx*4, +4
	} else {
		// If the source start point is higher than the destination start point, or equal height but to the left,
		// then we compose the rows in right-to-left, bottom-up order instead of left-to-right, top-down.
		d0 += (dy - 1) * dst.Stride
		s0 += (dy - 1) * src.Stride
		ddelta = -dst.Stride
		sdelta = -src.Stride
		i0, i1, idelta = (dx-1)*4, -4, -4
	}
	for ; dy > 0; dy-- {
		dpix := dst.Pix[d0:]
		spix := src.Pix[s0:]
		for i := i0; i != i1; i += idelta {
			s := spix[i : i+4 : i+4] // Small cap improves performance, see https://golang.org/issue/27857
			sr := uint32(s[0]) * 0x101
			sg := uint32(s[1]) * 0x101
			sb := uint32(s[2]) * 0x101
			sa := uint32(s[3]) * 0x101

			// The 0x101 is here for the same reason as in drawRGBA.
			a := (m - sa) * 0x101

			d := dpix[i : i+4 : i+4] // Small cap improves performance, see https://golang.org/issue/27857
			d[0] = uint8((uint32(d[0])*a/m + sr) >> 8)
			d[1] = uint8((uint32(d[1])*a/m + sg) >> 8)
			d[2] = uint8((uint32(d[2])*a/m + sb) >> 8)
			d[3] = uint8((uint32(d[3])*a/m + sa) >> 8)
		}
		d0 += ddelta
		s0 += sdelta
	}
}

func drawNRGBAOver(dst *image.RGBA, r image.Rectangle, src *image.NRGBA, sp image.Point) {
	i0 := (r.Min.X - dst.Rect.Min.X) * 4
	i1 := (r.Max.X - dst.Rect.Min.X) * 4
	si0 := (sp.X - src.Rect.Min.X) * 4
	yMax := r.Max.Y - dst.Rect.Min.Y

	y := r.Min.Y - dst.Rect.Min.Y
	sy := sp.Y - src.Rect.Min.Y
	for ; y != yMax; y, sy = y+1, sy+1 {
		dpix := dst.Pix[y*dst.Stride:]
		spix := src.Pix[sy*src.Stride:]

		for i, si := i0, si0; i < i1; i, si = i+4, si+4 {
			// Convert from non-premultiplied color to pre-multiplied color.
			s := spix[si : si+4 : si+4] // Small cap improves performance, see https://golang.org/issue/27857
			sa := uint32(s[3]) * 0x101
			sr := uint32(s[0]) * sa / 0xff
			sg := uint32(s[1]) * sa / 0xff
			sb := uint32(s[2]) * sa / 0xff

			d := dpix[i : i+4 : i+4] // Small cap improves performance, see https://golang.org/issue/27857
			dr := uint32(d[0])
			dg := uint32(d[1])
			db := uint32(d[2])
			da := uint32(d[3])

func panicBounds32(pc uintptr, regs *[16]int32) { // called from panicBounds on 32-bit archs
	f := findfunc(pc)
	v := pcdatavalue(f, abi.PCDATA_PanicBounds, pc-1)

	code, signed, xIsReg, yIsReg, xVal, yVal := abi.BoundsDecode(int(v))

	if code == abi.BoundsIndex {
		panicCheck1(pc, "index out of range")
	} else {
		panicCheck1(pc, "slice bounds out of range")
	}

	var e boundsError
	e.code = code
	e.signed = signed
	if xIsReg {
		if signed {
			e.x = int64(regs[xVal])
		} else {
			e.x = int64(uint32(regs[xVal]))
		}
	} else {
		e.x = int64(xVal)
	}
	if yIsReg {
		e.y = int(regs[yVal])
	} else {
		e.y = yVal
	}
	panic(e)
}
