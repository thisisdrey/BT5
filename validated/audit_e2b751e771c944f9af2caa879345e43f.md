No vulnerability found for this question.

The `TransactionStateContainer` enforces a strict capacity bound that is unaffected by attacker packet volume or source IP distribution. `SchedulerController::new` initializes the container with `container_capacity = TOTAL_BUFFERED_PACKETS` (100,000) via `R::Container::with_capacity(container_capacity)` [1](#0-0) , and the underlying `Slab` is pre-allocated to `capacity + EXTRA_CAPACITY` (a fixed +1 slot to hold one transient transaction between insertion and eviction), not to an unbounded size [2](#0-1) .

Every insertion path goes through `push_ids_into_queue`, which after insertion computes `num_dropped = id_to_transaction_state.len().saturating_sub(self.capacity)` and evicts that many lowest-priority entries from both the priority queue and the map [3](#0-2) . This is invoked in `handle_packet_batch_message` for every incoming sanitized transaction via `container.push_ids_into_queue(std::iter::once(priority_id))`, with the drop count tracked as `num_dropped_on_capacity` [4](#0-3) .

This exact scenario — flooding with more than the container's capacity worth of valid, sanitized, low-fee transactions — is already covered by an existing integration test, `test_receive_and_buffer_overfull`, which sends `3 * TEST_CONTAINER_CAPACITY` transactions and asserts `num_dropped_on_capacity > 0` while `verify_container(&mut container, TEST_CONTAINER_CAPACITY)` confirms the container never exceeds its fixed capacity [5](#0-4) . A similar bound is also verified for the eviction/nonce-dedup interaction in `test_view_push_ids_to_queue` [6](#0-5) .

Since the container size is hard-capped at `TOTAL_BUFFERED_PACKETS` regardless of how many distinct transactions or source IPs are used to flood it, and excess low-priority transactions are dropped rather than queued unboundedly, the described unbounded heap growth does not occur.

    fn test_view_push_ids_to_queue() {

        let reserved_addresses = HashSet::default();
        let packet_parser = |data, priority, cost| {
            let view =
                SanitizedTransactionView::try_new_sanitized(data, &sanitize_config()).unwrap();
            let view = RuntimeTransaction::<SanitizedTransactionView<_>>::try_new(
                view,
                MessageHash::Compute,
                None,
            )
            .unwrap();
            let view = RuntimeTransaction::<ResolvedTransactionView<_>>::try_new(
                view,
                None,
                &reserved_addresses,
            )
            .unwrap();

            TransactionState::new(view, MaxAge::MAX, priority, cost)
        };

        // Push 2 transactions into the queue so buffer is full.
        for priority in [4, 5] {
            let (transaction, _max_age, priority, cost) = test_transaction(priority);
            let packet = Packet::from_data(None, transaction.to_versioned_transaction()).unwrap();
            let data = Bytes::copy_from_slice(packet.data(..).unwrap());
            let id = container.insert_map_only(packet_parser(data, priority, cost));
            let priority_id = TransactionPriorityId::new(priority, id);
            assert_eq!(
                container.push_ids_into_queue(std::iter::once(priority_id)),
                0
            );
        }

        // Push 5 additional packets in. 5 should be dropped.
        for priority in [10, 11, 12, 1, 2] {
            let (transaction, _max_age, priority, cost) = test_transaction(priority);
            let packet = Packet::from_data(None, transaction.to_versioned_transaction()).unwrap();
            let data = Bytes::copy_from_slice(packet.data(..).unwrap());
            let id = container.insert_map_only(packet_parser(data, priority, cost));
            let priority_id = TransactionPriorityId::new(priority, id);
            assert_eq!(
                container.push_ids_into_queue(std::iter::once(priority_id)),
                1,
            );
        }
        assert_eq!(container.pop().unwrap().priority, 12);
        assert_eq!(container.pop().unwrap().priority, 11);
        assert!(container.pop().is_none());

        // Container now has no items in the queue, but still has 5 items in the map.
        // If we attempt to push additional transactions to the queue, they
        // are rejected regardless of their priority.
        let priority = u64::MAX;
        let (transaction, _max_age, priority, cost) = test_transaction(priority);
        let packet = Packet::from_data(None, transaction.to_versioned_transaction()).unwrap();
        let data = Bytes::copy_from_slice(packet.data(..).unwrap());
        let id = container.insert_map_only(packet_parser(data, priority, cost));
        let priority_id = TransactionPriorityId::new(priority, id);
        assert_eq!(
            container.push_ids_into_queue(std::iter::once(priority_id)),

        );
        assert!(container.pop().is_none());
