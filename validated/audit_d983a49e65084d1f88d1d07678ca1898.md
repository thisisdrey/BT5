The reported issue (OpenClaw's Plivo webhook replay-key using an unsorted/uncanonicalized URL while signature verification canonicalizes query order) is specific to that npm application's webhook handling logic. I searched the Go standard library repository for analogous patterns — signature/verification logic that canonicalizes input differently from a replay/cache-key derivation step — across:

- Module proxy caching (`src/cmd/go/internal/modfetch/cache.go`, `proxy.go`, `repo.go`), where cache keys and canonical versions are computed consistently before use [1](#0-0) 
- TLS/X.509 signature verification (`src/crypto/tls/auth.go`, `src/crypto/x509/x509.go`), where the signed bytes and verification path use the same canonical input [2](#0-1) [3](#0-2) 
- `net/url` query encode/parse and `net/http/httputil` reverse-proxy query cleaning, where `Values.Encode` sorts keys consistently and no replay/cache key is derived from an uncanonicalized raw query in a security-critical path [4](#0-3) [5](#0-4) 
- `cmd/go/internal/auth`'s credential cache, which keys on host+path only, explicitly excluding query, so it is not vulnerable to query-reordering bypass by design [6](#0-5) 

None of these production Go paths hash or key a replay/reuse-detection cache using a differently-canonicalized representation than the one used for signature/cryptographic verification, so the bug class does not reproduce in this repository.

				r = newLoggingRepo(r)
			}

// Encode encodes the values into “URL encoded” form
// ("bar=baz&foo=quux") sorted by key.
func (v Values) Encode() string {
	if len(v) == 0 {
		return ""
	}
	var buf strings.Builder
	// To minimize allocations, we eschew iterators and pre-size the slice in
	// which we collect v's keys.
	keys := make([]string, len(v))
	var i int
	for k := range v {
		keys[i] = k
		i++
	}
	slices.Sort(keys)
	for _, k := range keys {
		vs := v[k]
		keyEscaped := QueryEscape(k)
		for _, v := range vs {
			if buf.Len() > 0 {
				buf.WriteByte('&')
			}
			buf.WriteString(keyEscaped)
			buf.WriteByte('=')
			buf.WriteString(QueryEscape(v))
		}
	}
	return buf.String()
}

func cleanQueryParams(s string) string {
	reencode := func(s string) string {
		v, _ := url.ParseQuery(s)
		return v.Encode()
	}
	if urlmaxqueryparams.Value() != "" {
		// Always reencode when a non-default urlmaxqueryparams is set.
		return reencode(s)
	}
	if numParams := strings.Count(s, "&") + 1; numParams > defaultMaxParams {
		// Too many query parameters.
		return reencode(s)
	}
	for i := 0; i < len(s); {
		switch s[i] {
		case ';':
			return reencode(s)
		case '%':
			if i+2 >= len(s) || !ishex(s[i+1]) || !ishex(s[i+2]) {
				return reencode(s)
			}
			i += 3
		default:
			i++
		}
	}
	return s
}
