The closest analog in this Go standard library fork is the file-descriptor reference-counting mechanism in `internal/poll` (`fd_mutex.go`'s `incref`/`increfAndClose`/`decref`), which is precisely the defense Go uses to prevent the same use-after-free class described in CVE-2022-3977 (concurrent "close" vs. another operation on the same descriptor). [1](#0-0) [2](#0-1) 

Every FD operation (`Read`, `Write`, `Seek`, `Fchdir`, etc.) must go through `incref`/`decref` or `readLock`/`readUnlock`, and `Close` uses `increfAndClose`, which atomically marks the descriptor closed via CAS and only lets the final `decref` invoke `destroy()` (the actual `close(2)`/deallocation). [3](#0-2) [4](#0-3)  This is exactly the pattern that MCTP's DROPTAG-vs-close race lacked in the kernel, and Go's implementation already closes that race by design — there is no window where a concurrent misc/read/write operation can observe or use a descriptor that has already been torn down, because `mutexClosed`/ref-count bookkeeping in `fdMutex.state` gates every path.

The repository even contains regression tests specifically targeting this bug class and confirming the current code behaves correctly under concurrent close-vs-operation races: [5](#0-4) [6](#0-5) [7](#0-6) [8](#0-7) 

Additionally, this bug class requires two goroutines *within the same process* racing on the same `*os.File`/`*net.FD` — there is no unauthenticated/remote attacker input path (HTTP request, TLS handshake, module checksum, template, archive extraction) that reaches this code with attacker-controlled timing, as required by the scan rules. Since the primitive is already defended by the `fdMutex` reference-counting design and no externally-triggerable analog exists, this does not qualify as a valid finding.

// increfAndClose sets the state of mu to closed.
// It returns false if the file was already closed.
func (mu *fdMutex) increfAndClose() bool {
	for {
		old := atomic.LoadUint64(&mu.state)
		if old&mutexClosed != 0 {
			return false
		}
		// Mark as closed and acquire a reference.
		new := (old | mutexClosed) + mutexRef
		if new&mutexRefMask == 0 {
			panic(overflowMsg)
		}
		// Remove all read and write waiters.
		new &^= mutexRMask | mutexWMask
		if atomic.CompareAndSwapUint64(&mu.state, old, new) {
			// Wake all read and write waiters,
			// they will observe closed flag after wakeup.
			for old&mutexRMask != 0 {
				old -= mutexRWait
				runtime_Semrelease(&mu.rsema)
			}
			for old&mutexWMask != 0 {
				old -= mutexWWait
				runtime_Semrelease(&mu.wsema)
			}
			return true
		}
	}

	// so this must be executed before CloseFunc.

	fd.Sysfd = -1
	runtime_Semrelease(&fd.csema)
	return err

// Close closes the FD. The underlying file descriptor is closed by the
// destroy method when there are no remaining references.
func (fd *FD) Close() error {
	if !fd.fdmu.increfAndClose() {
		return errClosing(fd.isFile)
	}

	// Unblock any I/O.  Once it all unblocks and returns,
	// so that it cannot be referring to fd.sysfd anymore,
	// the final decref will close fd.sysfd. This should happen
	// fairly quickly, since all the I/O is non-blocking, and any
	// attempts to block in the pollDesc will return errClosing(fd.isFile).
	fd.pd.evict()

	// The call to decref will call destroy if there are no other
	// references.
	err := fd.decref()

	// Wait until the descriptor is closed. If this was the only
	// reference, it is already closed. Only wait if the file has
	// not been set to blocking mode, as otherwise any current I/O
	// may be blocking, and that would block the Close.
	// No need for an atomic read of isBlocking, increfAndClose means
	// we have exclusive access to fd.
	if fd.isBlocking == 0 {
		runtime_Semacquire(&fd.csema)
	}

	return err
}

	t.Parallel()
	for i := range 100 {
		name := pipeName()
		writer := newBytePipe(t, name, true)
		reader := newFileOverlapped(t, name, true)
		reader.Fd() // Use event-backed I/O.

		var wg sync.WaitGroup
		readDone := make(chan error, 1)
		wg.Go(func() {
			var buf [1]byte
			_, err := reader.Read(buf[:])
			readDone <- err
		})
		// Give Read a chance to acquire its FD reference, then race Close
		// against submission of the I/O request.
		time.Sleep(time.Nanosecond)
		closeDone := make(chan error, 1)
		wg.Go(func() { closeDone <- reader.Close() })
		select {
		case err := <-closeDone:
			if err != nil {
				t.Error(err)
			}
		case <-time.After(5 * time.Second):
			// Release the read even if Close missed its cancellation.
			writer.Close()
			wg.Wait()
			t.Fatalf("iteration %d: Close did not unblock Read", i)
		}
		wg.Wait()
		writer.Close()
		if err := <-readDone; !errors.Is(err, os.ErrClosed) {
			t.Fatalf("iteration %d: Read error = %v; want ErrClosed", i, err)
		}
	}
