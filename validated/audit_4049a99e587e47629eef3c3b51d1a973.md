The reported issue is specific to SurrealDB's query planner: field-level `SELECT` permissions, index-backed `ORDER BY` pushdown, and post-projection redaction of restricted fields. The repository `Lauraivanka/go--002` is a mirror of the Go standard library/toolchain source tree, not a database engine. Its `database/sql` and `database/sql/driver` packages [1](#0-0)  only define generic driver interfaces (`Rows`, `Queryer`, `QueryerContext`) that pass queries through to whatever backend driver is registered [2](#0-1) ; there is no query planner, no index-based sort pushdown, and no field-level permission/redaction model anywhere in this codebase that could exhibit the "ORDER BY leaks hidden field ordering" bug class. The closest structural analogs found — `slices.SortFunc` field-ordering logic in `encoding/json` [3](#0-2)  and struct field visibility/embedding resolution in `reflect` [4](#0-3)  — deal with compile-time/reflection field name shadowing, not runtime access-control permission checks on data values, so they are not a valid analog for this CWE-200 information-disclosure primitive.

	if w.visiting[t] {
		return
	}
	w.visiting[t] = true
	for i := 0; i < t.NumField(); i++ {
		f := t.Field(i)
		w.index = append(w.index, i)
		add := true
		if oldIndex, ok := w.byName[f.Name]; ok {
			old := &w.fields[oldIndex]
			if len(w.index) == len(old.Index) {
				// Fields with the same name at the same depth
				// cancel one another out. Set the field name
				// to empty to signify that has happened, and
				// there's no need to add this field.
				old.Name = ""
				add = false
			} else if len(w.index) < len(old.Index) {
				// The old field loses because it's deeper than the new one.
				old.Name = ""
			} else {
				// The old field wins because it's shallower than the new one.
				add = false
			}
		}

			// by the other appends.
