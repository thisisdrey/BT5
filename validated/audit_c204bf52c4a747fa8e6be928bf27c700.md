**Rationale:** Advertised remote addresses are always associated with the VPN identity that is cryptographically authenticated to have sent them, not with an attacker-chosen identity.

- `handleHostUpdateNotification` and `handleHostQueryReply` write into a `RemoteList` obtained via `lh.unlockedGetRemoteList(fromVpnAddrs)`/`certVpnAddr`, where `fromVpnAddrs` is derived from a decrypted, Noise-tunnel-authenticated packet whose identity comes from a CA-signed certificate — not from any attacker-supplied field [1](#0-0) [2](#0-1) .
- `handleHostQueryReply` additionally requires `lhh.lh.IsAnyLighthouseAddr(fromVpnAddrs)` before accepting reported addresses, so only an address that has completed a legitimate, cert-verified handshake as a configured lighthouse can inject entries [3](#0-2) .
- `handleHostUpdateNotification` also cross-checks that any self-reported `detailsVpnAddr` matches `fromVpnAddrs`, rejecting mismatched claims [4](#0-3) .
- All reported/advertised addresses pass through `shouldAdd`/`unlockedShouldAddV4`/`unlockedShouldAddV6`, which enforce `lighthouse.remote_allow_list`/`remote_allow_ranges` and exclude addresses inside the local VPN network table (`myVpnNetworksTable.Contains`) [5](#0-4) .
- `RemoteList` internally stores addresses per-owner key inside a cache map, but this cache is only reachable through the outer `RemoteList` instance that is already bound to the cert-verified identity (`lh.addrMap[addr]` keyed by that identity's own VPN addresses) [6](#0-5) [7](#0-6) .
- Since establishing an authenticated Noise tunnel or lighthouse identity requires a CA-signed certificate (explicitly excluded from the attacker's capabilities per the rules), an unprivileged attacker without a valid cert cannot forge `fromVpnAddrs`/`certVpnAddr` to inject a third-party address into another identity's `RemoteList`, and `NewLightHouseFromConfig`'s config-driven filters (`remote_allow_list`, `myVpnNetworksTable`) remain in effect regardless.

The invariant described in the question — advertised remotes only ever used for the identity that owns them — is already enforced through this chain of certificate-bound identity plus allow-list filtering, so the described exploit path (advertising a remote for a third party to make a victim send traffic there) is not reachable by an unprivileged attacker as scoped.

func (lh *LightHouse) unlockedGetRemoteList(allAddrs []netip.Addr) *RemoteList {
	// before we go and make a new remotelist, we need to make sure we don't have one for any of this set of vpnaddrs yet
	for i, addr := range allAddrs {
		am, ok := lh.addrMap[addr]
		if ok {
			if i != 0 {
				lh.addrMap[allAddrs[0]] = am
			}
			return am
		}
	}

	am := NewRemoteList(allAddrs, lh.shouldAdd)
	for _, addr := range allAddrs {
		lh.addrMap[addr] = am
	}
	return am
}

func (lhh *LightHouseHandler) handleHostQueryReply(n *NebulaMeta, fromVpnAddrs []netip.Addr) {
	if !lhh.lh.IsAnyLighthouseAddr(fromVpnAddrs) {
		return
	}

	certVpnAddr, _, err := n.Details.GetVpnAddrAndVersion()
	if err != nil {
		if lhh.l.Enabled(context.Background(), slog.LevelDebug) {
			lhh.l.Error("dropping malformed HostQueryReply",
				"error", err,
				"vpnAddrs", fromVpnAddrs,
			)
		}
		return
	}
	relays := n.Details.GetRelays()

	lhh.lh.Lock()
	am := lhh.lh.unlockedGetRemoteList([]netip.Addr{certVpnAddr})
	am.Lock()
	lhh.lh.Unlock()

	am.unlockedSetV4(fromVpnAddrs[0], certVpnAddr, n.Details.V4AddrPorts, lhh.lh.unlockedShouldAddV4)
	am.unlockedSetV6(fromVpnAddrs[0], certVpnAddr, n.Details.V6AddrPorts, lhh.lh.unlockedShouldAddV6)
	am.unlockedSetRelay(fromVpnAddrs[0], relays)

// RemoteList is a unifying concept for lighthouse servers and clients as well as hostinfos.
// It serves as a local cache of query replies, host update notifications, and locally learned addresses
type RemoteList struct {
	// Every interaction with internals requires a lock!
	sync.RWMutex

	// The full list of vpn addresses assigned to this host
	vpnAddrs []netip.Addr

	// A deduplicated set of underlay addresses. Any accessor should lock beforehand.
	addrs []netip.AddrPort

	// A set of relay addresses. VpnIp addresses that the remote identified as relays.
	relays []netip.Addr

	// These are maps to store v4 and v6 addresses per lighthouse
	// Map key is the vpnIp of the person that told us about this the cached entries underneath.
	// For learned addresses, this is the vpnIp that sent the packet
	cache map[netip.Addr]*cache

	hr *hostnamesResults

	// shouldAdd is a nillable function that decides if x should be added to addrs.
	shouldAdd func(vpnAddrs []netip.Addr, x netip.Addr) bool

	// This is a list of remotes that we have tried to handshake with and have returned from the wrong vpn ip.
	// They should not be tried again during a handshake
	badRemotes []netip.AddrPort

	// A flag that the cache may have changed and addrs needs to be rebuilt
	shouldRebuild bool
}
