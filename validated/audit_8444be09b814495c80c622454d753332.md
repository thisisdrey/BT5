The CVE-2021-4246 report concerns a PHP application (roxlukas/lmeve) that builds SQL queries from an untrusted `X-Forwarded-For` header. This bug class (string-concatenated SQL query construction from an HTTP header) has no analog in this repository, which is the Go language distribution itself (`Kohvert/go--017`) — not a web application with a login page or database backend.

Within this repo, the `database/sql` package consistently uses parameterized queries via placeholders and `driver.NamedValue`/`?` bind arguments rather than string concatenation, e.g. `db.QueryContext(ctx, "SELECT name FROM users WHERE age=?", age)` [1](#0-0) , `pool.QueryRowContext(ctx, "select p.name from people as p where p.id = :id;", sql.Named("id", id))` [2](#0-1) , and the core `db.QueryContext`/`db.ExecContext` entry points pass `args []any` through to the driver as `driver.NamedValue` rather than interpolating into the query string [3](#0-2) . There is no code path in `database/sql` (or elsewhere in this repo) that takes an HTTP header value and concatenates it into a SQL statement.

The only place `X-Forwarded-For` is handled in this repo is in `net/http/httputil`'s `ReverseProxy`, which appends/sets the header for proxying purposes — an unrelated concern to SQL construction, with no SQL sink involved [4](#0-3) .

Since the Go repository is the language toolchain/standard library and contains no application-level login page, session handling, or SQL query construction from request headers, there is no reachable production Go code path analogous to the reported PHP SQL injection.

// QueryContext executes a query that returns rows, typically a SELECT.
// The args are for any placeholder parameters in the query.
func (db *DB) QueryContext(ctx context.Context, query string, args ...any) (*Rows, error) {
	var rows *Rows
	var err error

	err = db.retry(func(strategy connReuseStrategy) error {
		rows, err = db.query(ctx, query, args, strategy)
		return err
	})

	return rows, err
}

// Query executes a query that returns rows, typically a SELECT.
// The args are for any placeholder parameters in the query.
//
// Query uses [context.Background] internally; to specify the context, use
// [DB.QueryContext].
func (db *DB) Query(query string, args ...any) (*Rows, error) {
	return db.QueryContext(context.Background(), query, args...)
}

func (db *DB) query(ctx context.Context, query string, args []any, strategy connReuseStrategy) (*Rows, error) {
	dc, err := db.conn(ctx, strategy)
	if err != nil {
		return nil, err
	}

	return db.queryDC(ctx, nil, dc, dc.releaseConn, query, args)
}
