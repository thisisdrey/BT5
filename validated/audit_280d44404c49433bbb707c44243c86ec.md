    fn verify(public_key: &Self::PublicKey, msg: &[u8], signature: &Self::Signature) -> bool {

        }

        cfg_select! {
            near => {
                ::near_sdk::env::ed25519_verify(
                    &signature.to_bytes(),
                    msg,
                    public_key.as_bytes(),
                )
            }
            _ => {{
                use ed25519_dalek::Verifier;

                public_key.verify(msg, signature).is_ok()
            }}
        }
    }
