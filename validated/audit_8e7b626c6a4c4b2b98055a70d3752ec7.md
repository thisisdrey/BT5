The reported CVE concerns a Python static-analysis tool (`picklescan`) failing to flag a specific dangerous call pattern (`numpy.f2py.crackfortran.myeval` reached through pickle's `REDUCE` opcode) as unsafe. This bug class is specific to pickle's opcode-based deserialization, which allows an attacker to name arbitrary importable callables and invoke them with attacker-controlled arguments purely from serialized data.

The closest Go analog is `encoding/gob`, which explicitly does not use this pattern. Gob decoding into an interface only proceeds by:
1. Looking up a concrete type that must have been explicitly pre-registered by the *program*, not the attacker, via `gob.Register`/`RegisterName` [1](#0-0) .
2. Dispatching to fixed, compile-time-defined interface methods (`GobDecode`, `UnmarshalBinary`, `UnmarshalText`) only if the concrete Go type implements them [2](#0-1) [3](#0-2) .
3. Never calls an attacker-named function by string lookup — there is no equivalent of pickle's `REDUCE`/`GLOBAL` opcode that resolves and invokes an arbitrary attacker-chosen callable by name.

Because gob has no "detect and reject dangerous global function" scanning step to bypass in the first place — untrusted data can never select the function to invoke, only supply bytes to a program-registered method — there's no equivalent "detection bypass" surface. This is a fundamentally different trust model, not a variant of the same bug class, so no genuine analog vulnerability exists in this repository.

// RegisterName is like [Register] but uses the provided name rather than the
// type's default.
func RegisterName(name string, value any) {
	if name == "" {
		// reserved for nil
		panic("attempt to register empty name")
	}

	ut := userType(reflect.TypeOf(value))

	// Check for incompatible duplicates. The name must refer to the
	// same user type, and vice versa.

	// Store the name and type provided by the user....
	if t, dup := nameToConcreteType.LoadOrStore(name, reflect.TypeOf(value)); dup && t != ut.user {
		panic(fmt.Sprintf("gob: registering duplicate types for %q: %s != %s", name, t, ut.user))
	}

	// but the flattened type in the type table, since that's what decode needs.
	if n, dup := concreteTypeToName.LoadOrStore(ut.base, name); dup && n != name {
		nameToConcreteType.Delete(name)
		panic(fmt.Sprintf("gob: registering duplicate names for %s: %q != %q", ut.user, n, name))
	}
}

// decodeGobDecoder decodes something implementing the GobDecoder interface.
// The data is encoded as a byte slice.
func (dec *Decoder) decodeGobDecoder(ut *userTypeInfo, state *decoderState, value reflect.Value) {
	// Read the bytes for the value.
	n, ok := state.getLength()
	if !ok {
		errorf("GobDecoder: length too large for buffer")
	}
	b := state.b.Bytes()
	if len(b) < n {
		errorf("GobDecoder: invalid data length %d: exceeds input size %d", n, len(b))
	}
	b = b[:n]
	state.b.Drop(n)
	var err error
	// We know it's one of these.
	switch ut.externalDec {
	case xGob:
		gobDecoder, _ := reflect.TypeAssert[GobDecoder](value)
		err = gobDecoder.GobDecode(b)
	case xBinary:
		binaryUnmarshaler, _ := reflect.TypeAssert[encoding.BinaryUnmarshaler](value)
		err = binaryUnmarshaler.UnmarshalBinary(b)
	case xText:
		textUnmarshaler, _ := reflect.TypeAssert[encoding.TextUnmarshaler](value)
		err = textUnmarshaler.UnmarshalText(b)
	}
	if err != nil {
		error_(err)
	}
}
