The `encoding/gob` decoder is structurally different from Java's Hot Rod deserialization primitive that the CVE describes. Java's `ObjectInputStream`-based deserialization can invoke arbitrary `readObject()` methods on classes on the classpath purely from a type name in the byte stream, enabling gadget-chain RCE. Go's `gob.Decoder.DecodeValue` requires the caller to supply a concrete destination value/pointer for `Decode`, and for interface values it only accepts concrete types that have been explicitly registered via `gob.Register` ahead of time [1](#0-0) . There's no mechanism for attacker-supplied wire data to select or instantiate an arbitrary, unregistered type or invoke unexpected code — decoding is confined to whatever `GobDecode`/`UnmarshalBinary`/`UnmarshalText` method the pre-registered destination type already implements [2](#0-1) .

Additionally, the package's own documentation already discloses and accepts this class of risk, explicitly stating it is "not designed to be hardened against adversarial inputs" and is "outside the scope" of the Go security policy, directing callers such as `net/rpc` to take care when decoding gob data from untrusted sources [3](#0-2) [4](#0-3) . `net/rpc`'s `gobServerCodec.ReadRequestBody`/`ReadRequestHeader` decode into server-controlled, pre-declared argument types, not attacker-chosen ones [5](#0-4) . This is a known, documented, and accepted design tradeoff rather than a fixable defect analogous to the Infinispan CVE, so it does not qualify as a new finding under the given rules.

func (dec *Decoder) DecodeValue(v reflect.Value) error {
	if v.IsValid() {
		if v.Kind() == reflect.Pointer && !v.IsNil() {
			// That's okay, we'll store through the pointer.
		} else if !v.CanSet() {
			return errors.New("gob: DecodeValue of unassignable value")
		}
	}
	// Make sure we're single-threaded through here.
	dec.mutex.Lock()
	defer dec.mutex.Unlock()

	dec.buf.Reset() // In case data lingers from previous invocation.
	dec.err = nil
	id := dec.decodeTypeSequence(false)
	if dec.err == nil {
		dec.decodeValue(id, v)
	}
	return dec.err

// decodeGobDecoder decodes something implementing the GobDecoder interface.
// The data is encoded as a byte slice.
func (dec *Decoder) decodeGobDecoder(ut *userTypeInfo, state *decoderState, value reflect.Value) {
	// Read the bytes for the value.
	n, ok := state.getLength()
	if !ok {
		errorf("GobDecoder: length too large for buffer")
	}
	b := state.b.Bytes()
	if len(b) < n {
		errorf("GobDecoder: invalid data length %d: exceeds input size %d", n, len(b))
	}
	b = b[:n]
	state.b.Drop(n)
	var err error
	// We know it's one of these.
	switch ut.externalDec {
	case xGob:
		gobDecoder, _ := reflect.TypeAssert[GobDecoder](value)
		err = gobDecoder.GobDecode(b)
	case xBinary:
		binaryUnmarshaler, _ := reflect.TypeAssert[encoding.BinaryUnmarshaler](value)
		err = binaryUnmarshaler.UnmarshalBinary(b)
	case xText:
		textUnmarshaler, _ := reflect.TypeAssert[encoding.TextUnmarshaler](value)
		err = textUnmarshaler.UnmarshalText(b)
	}
	if err != nil {
		error_(err)
	}
}
